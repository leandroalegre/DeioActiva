# =============================================================================
# .htaccess — deioactiva.sistemascloud.ar
# Ubicacion en servidor: /home/sistemacbahost/deioactiva.sistemascloud.ar/.htaccess
#   (raiz del subdominio, junto al build del frontend)
#
# Funcion doble:
#   1. Proxy /api/ -> NestJS corriendo en 127.0.0.1:3001 (PM2: deioactiva-api)
#      via RewriteRule [P] (NO ProxyPass directo: ese hosting no lo permite
#      en .htaccess, devuelve 500 - confirmado a mano en el deploy real).
#   2. React SPA routing -> todas las rutas no-archivo van a index.html
# =============================================================================

Options -MultiViews
RewriteEngine On

# 1. PROXY: /api/ -> backend NestJS (puerto 3001), via RewriteRule [P]
RewriteCond %{REQUEST_URI} ^/api/
RewriteRule ^api/(.*)$ http://127.0.0.1:3001/api/$1 [P,L]

# 2. SPA ROUTING: React Router (todo lo demas -> index.html)
RewriteCond %{REQUEST_URI} !^/api/
RewriteCond %{REQUEST_FILENAME} !-f
RewriteCond %{REQUEST_FILENAME} !-d
RewriteRule ^ /index.html [L]

# 3. Seguridad basica
<FilesMatch "\.(env|sh|json|lock|log)$">
  Order allow,deny
  Deny from all
</FilesMatch>

<IfModule mod_headers.c>
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>
